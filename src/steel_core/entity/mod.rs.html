24/// Global counter for allocating unique entity IDs.
25///
26/// Mirrors vanilla's `Entity.ENTITY_COUNTER`. Each new entity increments this
27/// counter to get a unique network ID. Starts at 1 (0 is reserved).
28static ENTITY_COUNTER: AtomicI32 = AtomicI32::new(1);
29
30/// Allocates a new unique entity ID.
31///
32/// This is the primary way to get entity IDs for spawning entities.
33/// Thread-safe and lock-free.
34#[must_use]
35pub fn next_entity_id() -> i32 {
36    ENTITY_COUNTER.fetch_add(1, Ordering::Relaxed)
37}

66/// A trait for entities.
67///
68/// This trait provides the core functionality for entities.
69/// It's based on Minecraft's `Entity` class.
70///
71/// # Using `EntityBase`
72///
73/// Entities that embed [`EntityBase`] can implement `base()` to get default
74/// implementations for common methods like `id()`, `uuid()`, `position()`, etc.
75///
76/// ```ignore
77/// impl Entity for MyEntity {
78///     fn base(&self) -> Option<&EntityBase> { Some(&self.base) }
79///     fn entity_type(&self) -> EntityTypeRef { vanilla_entities::MY_ENTITY }
80///     fn bounding_box(&self) -> AABBd { /* ... */ }
81///     // All other common methods use defaults from EntityBase!
82/// }
83/// ```
84pub trait Entity: Send + Sync {
85    /// Returns a reference to the entity's base fields, if available.
86    ///
87    /// Implement this to get default implementations for common methods.
88    /// Returns `None` by default (for entities like Player that don't use `EntityBase`).
89    fn base(&self) -> Option<&EntityBase> {
90        None
91    }

93    /// Gets the entity type containing tracking range, dimensions, etc.
94    fn entity_type(&self) -> EntityTypeRef;

97    fn id(&self) -> i32 {
98        self.base().map_or(0, EntityBase::id)

102    fn uuid(&self) -> Uuid {
103        self.base().map_or(Uuid::nil(), EntityBase::uuid)

106    /// Gets the entity's current position.
107    fn position(&self) -> DVec3 {
108        self.base()
109            .map_or(DVec3::new(0.0, 0.0, 0.0), EntityBase::position)
110    }

112    /// Gets the entity's bounding box for collision queries.

116    ///
117    /// Use `self.level()` to access the world for physics, block queries, etc.
118    /// The caller (`EntityStorage`) handles base tick logic like dirty data sync.

122    ///
123    /// Called every tick by `EntityStorage` after `tick()`, mirrors vanilla's

228    // === Physics Helper Methods ===
229    // These mirror vanilla's Entity class methods.
230
231    /// Gets the default gravity for this entity type.
232    ///
233    /// Override this to specify entity-specific gravity.
234    /// Vanilla values: `ItemEntity` = 0.04, `LivingEntity` = 0.08
235    fn get_default_gravity(&self) -> f64 {
236        0.0
237    }
238
239    /// Returns true if gravity is disabled for this entity.
240    ///
241    /// Override to read from entity data's `no_gravity` field.
242    fn is_no_gravity(&self) -> bool {
243        false
244    }
245
246    /// Gets the current gravity value.
247    ///
248    /// Returns 0 if `no_gravity` is set, otherwise returns `get_default_gravity()`.
249    fn get_gravity(&self) -> f64 {
250        if self.is_no_gravity() {
251            0.0
252        } else {
253            self.get_default_gravity()
254        }
255    }
256
257    /// Applies gravity to the entity's velocity.
258    ///
259    /// Mirrors vanilla's `Entity.applyGravity()`.
260    fn apply_gravity(&self) {
261        let gravity = self.get_gravity();
262        if gravity != 0.0 {
263            let mut vel = self.velocity();
264            vel.y -= gravity;
265            self.set_velocity(vel);
266        }
267    }

269    /// Moves the entity with collision detection.
270    ///
271    /// Mirrors vanilla's `Entity.move(MoverType, Vec3)`.
272    /// Updates position, `on_ground`, velocity (on collision), and returns collision info.
273    fn do_move(&self, mover_type: MoverType) -> Option<MoveResult> {
274        let world = self.level()?;
275        let velocity = self.velocity();
276
277        // Build physics state
278        let mut physics_state = EntityPhysicsState::new(self.position(), self.entity_type());
279        physics_state.velocity = velocity;
280        physics_state.on_ground = self.on_ground();
281        // Most entities don't step up; override for entities that do
282        physics_state.max_up_step = 0.0;
283        physics_state.is_crouching = false;
284
285        // Perform collision detection and movement
286        let collision_world = WorldCollisionProvider::new(&world);
287        let result = move_entity(&physics_state, velocity, mover_type, &collision_world);
288
289        // Update entity state
290        self.set_position(result.final_position);
291        self.set_on_ground(result.on_ground);
292
293        // Vanilla: Entity.move() zeros velocity components on collision.
294        // Horizontal collision zeros X/Z individually based on which axis collided.
295        // Vertical collision calls Block.updateEntityMovementAfterFallOn which by default zeros Y.
296        // (vanilla: Entity.move lines 776-785)
297        // TODO: Support block-specific behavior (slime bounce, etc.)
298        if result.horizontal_collision {
299            let vel = self.velocity();
300            self.set_velocity(DVec3::new(
301                if result.x_collision { 0.0 } else { vel.x },
302                vel.y,
303                if result.z_collision { 0.0 } else { vel.z },
304            ));
305        }
306        if result.vertical_collision {
307            // Default Block.updateEntityMovementAfterFallOn behavior: zero Y velocity
308            let vel = self.velocity();
309            self.set_velocity(DVec3::new(vel.x, 0.0, vel.z));
310        }
311
312        Some(result)
313    }

315    /// Spawns an item at this entity's location.
316    ///
317    /// Mirrors vanilla's `Entity.spawnAtLocation()`. The item spawns at the
318    /// entity's position with the given Y offset and has a default pickup delay.
319    ///
320    /// Returns `None` if the item stack is empty or the entity has no world.
321    fn spawn_at_location(
322        &self,
323        item: ItemStack,
324        y_offset: f64,
325    ) -> Option<Arc<entities::ItemEntity>> {
326        let world = self.level()?;
327        let pos = self.position();
328        world.spawn_item(DVec3::new(pos.x, pos.y + y_offset, pos.z), item)
329    }

351    // === Tick Tracking ===
352    // These methods prevent double-ticking when an entity moves between chunks

354
355    /// Checks if this entity was already ticked during the given server tick.
356    ///
357    /// This prevents double-ticking when an entity moves to a different chunk
358    /// during its tick, and that chunk gets ticked later in the same server tick.
359    ///
360    /// Returns `true` if already ticked this tick, `false` otherwise.
361    fn was_ticked_this_tick(&self, server_tick: i32) -> bool {
362        self.base()
363            .is_some_and(|b| b.was_ticked_this_tick(server_tick))
364    }

367    ///
368    /// Called by `EntityStorage::tick()` before ticking an entity.

371            base.mark_ticked(server_tick);
372        }
373    }

389/// A trait for living entities that can take damage, heal, and die.
390///
391/// This trait provides the core functionality for entities that have health,
392/// can be damaged, and can die. It's based on Minecraft's `LivingEntity` class.
393///
394/// **Note:** All methods take `&self` (not `&mut self`) because living entities
395/// are shared via `Arc` and use interior mutability (atomics, `SyncMutex`, etc.).
396pub trait LivingEntity: Entity {
397    /// Gets the current health of the entity.
398    fn get_health(&self) -> f32;
399
400    /// Sets the health of the entity, clamped between 0 and max health.
401    fn set_health(&self, health: f32);
402
403    /// Gets the maximum health of the entity.
404    fn get_max_health(&self) -> f32;
405
406    /// Heals the entity by the specified amount.
407    fn heal(&self, amount: f32) {
408        let current_health = self.get_health();
409        if current_health > 0.0 {
410            self.set_health(current_health + amount);
411        }
412    }
413
414    /// Returns true if the entity is dead or dying (health <= 0).
415    fn is_dead_or_dying(&self) -> bool {
416        self.get_health() <= 0.0
417    }
418
419    /// Returns true if the entity is alive (health > 0).
420    fn is_alive(&self) -> bool {
421        !self.is_dead_or_dying()
422    }
423
424    /// Returns a reference to the shared [`LivingEntityBase`] that holds
425    /// `dead`, `invulnerable_time`, and `last_hurt`.
426    fn living_base(&self) -> &SyncMutex<LivingEntityBase>;
427
428    /// Gets the absorption amount (extra health from effects like absorption).
429    fn get_absorption_amount(&self) -> f32;
430
431    /// Sets the absorption amount.
432    fn set_absorption_amount(&self, amount: f32);
433
434    /// Gets the entity's armor value.
435    fn get_armor_value(&self) -> i32;
436
437    /// Checks if the entity can be affected by potions.
438    fn is_affected_by_potions(&self) -> bool {
439        true
440    }
441
442    /// Checks if the entity is attackable.
443    fn attackable(&self) -> bool {
444        true
445    }
446
447    /// Checks if the entity is currently using an item.
448    fn is_using_item(&self) -> bool {
449        false
450    }
451
452    /// Checks if the entity is blocking with a shield or similar item.
453    fn is_blocking(&self) -> bool {
454        false
455    }
456
457    /// Checks if the entity is fall flying (using elytra).
458    fn is_fall_flying(&self) -> bool {
459        false
460    }
461
462    /// Checks if the entity is sleeping.
463    fn is_sleeping(&self) -> bool {
464        false
465    }
466
467    /// Stops the entity from sleeping.
468    fn stop_sleeping(&self) {}
469
470    /// Checks if the entity is sprinting.
471    fn is_sprinting(&self) -> bool {
472        false
473    }
474
475    /// Sets whether the entity is sprinting.
476    fn set_sprinting(&self, sprinting: bool);
477
478    /// Gets the entity's speed attribute value.
479    fn get_speed(&self) -> f32;
480
481    /// Sets the entity's speed.
482    fn set_speed(&self, speed: f32);
